    function calculateMacros() {
      const weight = parseFloat(document.getElementById('calc-weight').value);
      const height = parseFloat(document.getElementById('calc-height').value);
      const age = parseFloat(document.getElementById('calc-age').value);
      const activity = parseFloat(document.getElementById('calc-activity').value);
      const goal = document.getElementById('calc-goal').value;

      if (!weight || !height || !age) {
        alert("Bitte fülle alle Eingabefelder aus!");
        return;
      }

      // BMR (Mifflin-St Jeor Formel für Männer)
      let bmr = (10 * weight) + (6.25 * height) - (5 * age) + 5;
      let tdee = bmr * activity;

      if (goal === 'cut') tdee -= 500;
      if (goal === 'bulk') tdee += 300;

      const targetCals = Math.round(tdee);
      const protein = Math.round(weight * 2.0); // 2g pro kg
      const fats = Math.round(weight * 0.9);    // 0.9g pro kg
      const carbCalories = targetCals - ((protein * 4) + (fats * 9));
      const carbs = Math.max(0, Math.round(carbCalories / 4));

      document.getElementById('res-calories').innerText = `${targetCals} kcal`;
      document.getElementById('res-protein').innerText = `${protein}g`;
      document.getElementById('res-carbs').innerText = `${carbs}g`;
      document.getElementById('res-fats').innerText = `${fats}g`;

      document.getElementById('calc-result').classList.remove('hidden');
    }

    const hairstyles = {

      },
      square: {
        title: "Eckiges Gesicht (Square)",
        desc: "Markanter Kiefer! Halte die Seiten kurz (Skin Fade / Low Fade) und oben etwas mehr Volumen. Empfehlungen: Buzz Cut, Crew Cut, Slicked Back oder Pompadour."
      },

      }

    function updateHabit() {
      const checks = document.querySelectorAll('.habit-check');
